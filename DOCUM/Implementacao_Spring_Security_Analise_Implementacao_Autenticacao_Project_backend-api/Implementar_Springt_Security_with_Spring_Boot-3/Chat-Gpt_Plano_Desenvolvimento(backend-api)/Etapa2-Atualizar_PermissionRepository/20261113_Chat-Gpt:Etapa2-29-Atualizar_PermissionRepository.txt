Para lembrar dos codigos HTTP combinados com os verbos do HTTP:

• 200 OK: A requisição foi bem-sucedida. O significado do sucesso depende do método HTTP utilizado
  (ex: em um GET, os dados solicitados são retornados no corpo da resposta).

• 204 No Content: A requisição foi processada com sucesso, mas não há conteúdo para retornar no
  corpo da resposta (altamente utilizado em requisições DELETE ou PUT/PATCH quando não há necessidade
  de devolver o objeto atualizado).

• 401 Unauthorized: O cliente precisa se autenticar para obter a resposta solicitada 
  (falta de token ou credenciais inválidas).

• 403 Forbidden: O cliente foi autenticado, mas não tem direitos de acesso ao recurso 
  (permissão insuficiente).

• 409 Conflict: A requisição não pôde ser concluída devido a um conflito com o estado atual do 
  servidor (ex: tentar cadastrar um usuário com um e-mail que já existe).

Etapa 4 — Criar PermissionService.java

Caminho:

`src/main/java/com/julyturadm/service/PermissionService.java`

Conteúdo completo:

```

package com.julyturadm.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.julyturadm.common.exception.BusinessException;
import com.julyturadm.common.exception.ResourceNotFoundException;
import com.julyturadm.domain.entity.Permission;
import com.julyturadm.domain.repository.PermissionRepository;
import com.julyturadm.domain.repository.RolePermissionRepository;
import com.julyturadm.dto.request.PermissionCreateRequest;
import com.julyturadm.dto.request.PermissionUpdateRequest;
import com.julyturadm.dto.response.PermissionResponse;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class PermissionService {

    private final PermissionRepository permissionRepository;
    private final RolePermissionRepository rolePermissionRepository;

    public PermissionResponse createPermission(PermissionCreateRequest request) {

        if (permissionRepository.existsByName(request.name())) {
            throw new BusinessException("Já existe uma permissão com esse nome.");
        }

        Permission permission = Permission.create(request.name());

        Permission savedPermission = permissionRepository.save(permission);

        return toResponse(savedPermission);
    }

    @Transactional(readOnly = true)
    public PermissionResponse findPermissionById(Long id) {

        Permission permission = findPermissionOrThrow(id);

        return toResponse(permission);
    }

    @Transactional(readOnly = true)
    public List<PermissionResponse> findAllPermissions() {

        return permissionRepository.findAll()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    public PermissionResponse updatePermission(
            Long id,
            PermissionUpdateRequest request) {

        Permission permission = findPermissionOrThrow(id);

        if (permissionRepository.existsByNameAndIdNot(request.name(), id)) {
            throw new BusinessException("Já existe outra permissão com esse nome.");
        }

        permission.setName(request.name());

        Permission updatedPermission = permissionRepository.save(permission);

        return toResponse(updatedPermission);
    }

    public void deletePermission(Long id) {

        Permission permission = findPermissionOrThrow(id);

        if (rolePermissionRepository.existsByPermission_Id(id)) {
            throw new BusinessException(
                    "Não é possível excluir a permissão: ela está associada a perfis.");
        }

        permissionRepository.delete(permission);
    }

    private Permission findPermissionOrThrow(Long id) {

        return permissionRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Permissão não encontrada: " + id));
    }

    private PermissionResponse toResponse(Permission permission) {

        return new PermissionResponse(
                permission.getId(),
                permission.getName(),
                permission.getCreatedAt(),
                permission.getUpdatedAt());
    }
}

```

Regras implementadas

- Não permite criar permissões com nomes duplicados.
- Não permite atualizar para o nome de outra permissão existente.
- Permite manter o próprio nome durante uma atualização.
- Retorna 404 quando a permissão não existe.
- Impede excluir permissões associadas a perfis.
- Não exclui associações automaticamente.

Importante: não estamos alterando o catálogo de permissões automaticamente nem modificando as
associações `roles_permissions`.

package com.julyturadm.controller;

import java.util.List;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.web.bind.annotation.*;

import com.julyturadm.dto.request.PermissionCreateRequest;
import com.julyturadm.dto.request.PermissionUpdateRequest;
import com.julyturadm.dto.response.PermissionResponse;
import com.julyturadm.service.PermissionService;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;

@RestController
@RequestMapping("/api/permissions")
@RequiredArgsConstructor
public class PermissionController {

    private final PermissionService permissionService;

    @PostMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<PermissionResponse> createPermission(
            @Valid @RequestBody PermissionCreateRequest request) {

        PermissionResponse response =
                permissionService.createPermission(request);

        return ResponseEntity.status(HttpStatus.CREATED).body(response);
    }

    @GetMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<PermissionResponse> findPermissionById(
            @PathVariable Long id) {

        return ResponseEntity.ok(
                permissionService.findPermissionById(id));
    }

    @GetMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<List<PermissionResponse>> findAllPermissions() {

        return ResponseEntity.ok(
                permissionService.findAllPermissions());
    }

    @PutMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<PermissionResponse> updatePermission(
            @PathVariable Long id,
            @Valid @RequestBody PermissionUpdateRequest request) {

        return ResponseEntity.ok(
                permissionService.updatePermission(id, request));
    }

    @DeleteMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<Void> deletePermission(
            @PathVariable Long id) {

        permissionService.deletePermission(id);

        return ResponseEntity.noContent().build();
    }
}

import java.util.List;

import com.julyturadm.dto.request.PermissionUpdateRequest;
import com.julyturadm.dto.response.PermissionResponse;
import com.julyturadm.service.PermissionService;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;

@RestController
@RequestMapping("/api/permissions")
@RequiredArgsConstructor
public class PermissionController {

    private final PermissionService permissionService;

    @PostMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<PermissionResponse> createPermission(
            @Valid @RequestBody PermissionCreateRequest request) {

        PermissionResponse response =
                permissionService.createPermission(request);

        return ResponseEntity.status(HttpStatus.CREATED).body(response);
    }

    @GetMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<PermissionResponse> findPermissionById(
            @PathVariable Long id) {

        return ResponseEntity.ok(
                permissionService.findPermissionById(id));
    }

    @GetMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<List<PermissionResponse>> findAllPermissions() {

        return ResponseEntity.ok(
                permissionService.findAllPermissions());
    }

    @PutMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<PermissionResponse> updatePermission(
            @PathVariable Long id,
            @Valid @RequestBody PermissionUpdateRequest request) {

        return ResponseEntity.ok(
                permissionService.updatePermission(id, request));
    }

    @DeleteMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<Void> deletePermission(
            @PathVariable Long id) {

        permissionService.deletePermission(id);

        return ResponseEntity.noContent().build();
    }
}

package com.julyturadm.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.julyturadm.common.exception.BusinessException;
import com.julyturadm.common.exception.ResourceNotFoundException;
import com.julyturadm.domain.entity.Permission;
import com.julyturadm.domain.repository.PermissionRepository;
import com.julyturadm.domain.repository.RolePermissionRepository;
import com.julyturadm.dto.request.PermissionCreateRequest;
import com.julyturadm.dto.request.PermissionUpdateRequest;
import com.julyturadm.dto.response.PermissionResponse;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class PermissionService {

    private final PermissionRepository permissionRepository;
    private final RolePermissionRepository rolePermissionRepository;

    public PermissionResponse createPermission(PermissionCreateRequest request) {

        if (permissionRepository.existsByName(request.name())) {
            throw new BusinessException("Já existe uma permissão com esse nome.");
        }

        Permission permission = Permission.create(request.name());

        Permission savedPermission = permissionRepository.save(permission);

        return toResponse(savedPermission);
    }

    @Transactional(readOnly = true)
    public PermissionResponse findPermissionById(Long id) {

        Permission permission = findPermissionOrThrow(id);

        return toResponse(permission);
    }

    @Transactional(readOnly = true)
    public List<PermissionResponse> findAllPermissions() {

        return permissionRepository.findAll()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    public PermissionResponse updatePermission(
            Long id,
            PermissionUpdateRequest request) {

        Permission permission = findPermissionOrThrow(id);

        if (permissionRepository.existsByNameAndIdNot(request.name(), id)) {
            throw new BusinessException("Já existe outra permissão com esse nome.");
        }

        permission.setName(request.name());

        Permission updatedPermission = permissionRepository.save(permission);

        return toResponse(updatedPermission);
    }

    public void deletePermission(Long id) {

        Permission permission = findPermissionOrThrow(id);

        if (rolePermissionRepository.existsByPermission_Id(id)) {
            throw new BusinessException(
                    "Não é possível excluir a permissão: ela está associada a perfis.");
        }

        permissionRepository.delete(permission);
    }

    private Permission findPermissionOrThrow(Long id) {

        return permissionRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Permissão não encontrada: " + id));
    }

    private PermissionResponse toResponse(Permission permission) {

        return new PermissionResponse(
                permission.getId(),
                permission.getName(),
                permission.getCreatedAt(),
                permission.getUpdatedAt());
    }
}
