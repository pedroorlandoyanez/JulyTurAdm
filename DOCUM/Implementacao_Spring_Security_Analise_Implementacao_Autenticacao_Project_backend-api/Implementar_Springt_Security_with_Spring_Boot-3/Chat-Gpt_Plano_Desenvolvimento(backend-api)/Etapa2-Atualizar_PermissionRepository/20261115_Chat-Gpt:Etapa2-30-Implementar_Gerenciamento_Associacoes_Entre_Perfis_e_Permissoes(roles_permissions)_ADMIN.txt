Para lembrar dos codigos HTTP combinados com os verbos do HTTP:

• 200 OK: A requisição foi bem-sucedida. O significado do sucesso depende do método HTTP utilizado
  (ex: em um GET, os dados solicitados são retornados no corpo da resposta).

• 204 No Content: A requisição foi processada com sucesso, mas não há conteúdo para retornar no
  corpo da resposta (altamente utilizado em requisições DELETE ou PUT/PATCH quando não há necessidade
  de devolver o objeto atualizado).

• 401 Unauthorized: O cliente precisa se autenticar para obter a resposta solicitada 
  (falta de token ou credenciais inválidas).

• 403 Forbidden: O cliente foi autenticado, mas não tem direitos de acesso ao recurso 
  (permissão insuficiente).

• 409 Conflict: A requisição não pôde ser concluída devido a um conflito com o estado atual do 
  servidor (ex: tentar cadastrar um usuário com um e-mail que já existe).

package com.julyturadm.domain.entity;

import com.julyturadm.domain.entity.id.RolePermissionId;
import jakarta.persistence.*;
import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

@Entity
@Table(name = "roles_permissions")
@Getter
@Setter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class RolePermission {

    // - `RolePermissionId` recebe os IDs dos dois registros.
    @EmbeddedId
    private RolePermissionId id;

    // ---
    // - `setRole()` estabelece o relacionamento com o perfil.
    // - `@MapsId` mantém os campos da chave composta sincronizados com os
    //    relacionamentos no momento da persistência.
    // ---
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @MapsId("roleId")
    @JoinColumn(name = "role_id", nullable = false, foreignKey = @ForeignKey(name = "FK_rp_role"))
    private Role role;

    // ---
    // - `setPermission()` estabelece o relacionamento com a permissão.
    // - `@MapsId` mantém os campos da chave composta sincronizados com os
    //    relacionamentos no momento da persistência.
    // ---
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @MapsId("permissionId")
    @JoinColumn(name = "permission_id", nullable = false, foreignKey = @ForeignKey(name = "FK_rp_permission"))
    private Permission permission;
       
    // ---
    // Esse método permite criar uma role e uma permissao sem expor publicamente o 
    // construtor da entidade.
    // ---    
    public static RolePermission create(
            Role role,
            Permission permission) {

        RolePermission rolePermission = new RolePermission();

        rolePermission.setId(
                new RolePermissionId(
                        role.getId(),
                        permission.getId()));

        rolePermission.setRole(role);
        rolePermission.setPermission(permission);

        return rolePermission;
    }

}

package com.julyturadm.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.julyturadm.common.exception.BusinessException;
import com.julyturadm.common.exception.ResourceNotFoundException;
import com.julyturadm.domain.entity.Permission;
import com.julyturadm.domain.entity.Role;
import com.julyturadm.domain.entity.RolePermission;
import com.julyturadm.domain.repository.PermissionRepository;
import com.julyturadm.domain.repository.RolePermissionRepository;
import com.julyturadm.domain.repository.RoleRepository;
import com.julyturadm.dto.response.RolePermissionResponse;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class RolePermissionService {

    private final RoleRepository roleRepository;
    private final PermissionRepository permissionRepository;
    private final RolePermissionRepository rolePermissionRepository;

    @Transactional(readOnly = true)
    public List<RolePermissionResponse> findPermissionsByRoleId(Long roleId) {

        Role role = findRoleOrThrow(roleId);

        return rolePermissionRepository.findAllByRole_Id(roleId)
                .stream()
                .map(rolePermission -> toResponse(role, rolePermission.getPermission()))
                .toList();
    }

    public void assignPermissionToRole(Long roleId, Long permissionId) {

        Role role = findRoleOrThrow(roleId);

        Permission permission = findPermissionOrThrow(permissionId);

        if (rolePermissionRepository.existsByRole_IdAndPermission_Id(
                roleId, permissionId)) {

            throw new BusinessException(
                    "Essa permissão já está associada ao perfil.");
        }

        RolePermission rolePermission =
                RolePermission.create(role, permission);

        rolePermissionRepository.save(rolePermission);
    }

    public void removePermissionFromRole(Long roleId, Long permissionId) {

        Role role = findRoleOrThrow(roleId);

        Permission permission = findPermissionOrThrow(permissionId);

        RolePermissionId id =
                new RolePermissionId(roleId, permissionId);

        RolePermission rolePermission = rolePermissionRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Associação entre perfil e permissão não encontrada."));

        rolePermissionRepository.delete(rolePermission);
    }

    private Role findRoleOrThrow(Long roleId) {

        return roleRepository.findById(roleId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Perfil não encontrado: " + roleId));
    }

    private Permission findPermissionOrThrow(Long permissionId) {

        return permissionRepository.findById(permissionId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Permissão não encontrada: " + permissionId));
    }

    private RolePermissionResponse toResponse(
            Role role,
            Permission permission) {

        return new RolePermissionResponse(
                role.getId(),
                role.getName(),
                permission.getId(),
                permission.getName());
    }
}

package com.julyturadm.controller;

import java.util.List;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.web.bind.annotation.*;

import com.julyturadm.dto.response.RolePermissionResponse;
import com.julyturadm.service.RolePermissionService;

import lombok.RequiredArgsConstructor;

@RestController
@RequestMapping("/api/roles/{roleId}/permissions")
@RequiredArgsConstructor
public class RolePermissionController {

    private final RolePermissionService rolePermissionService;

    @GetMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<List<RolePermissionResponse>> findPermissionsByRoleId(
            @PathVariable Long roleId) {

        return ResponseEntity.ok(
                rolePermissionService.findPermissionsByRoleId(roleId));
    }

    @PutMapping("/{permissionId}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<Void> assignPermissionToRole(
            @PathVariable Long roleId,
            @PathVariable Long permissionId) {

        rolePermissionService.assignPermissionToRole(roleId, permissionId);

        return ResponseEntity.status(HttpStatus.CREATED).build();
    }

    @DeleteMapping("/{permissionId}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<Void> removePermissionFromRole(
            @PathVariable Long roleId,
            @PathVariable Long permissionId) {

        rolePermissionService.removePermissionFromRole(roleId, permissionId);

        return ResponseEntity.noContent().build();
    }
}
