Para lembrar dos codigos HTTP combinados com os verbos do HTTP:

• 200 OK: A requisição foi bem-sucedida. O significado do sucesso depende do método HTTP utilizado
  (ex: em um GET, os dados solicitados são retornados no corpo da resposta).

• 204 No Content: A requisição foi processada com sucesso, mas não há conteúdo para retornar no
  corpo da resposta (altamente utilizado em requisições DELETE ou PUT/PATCH quando não há necessidade
  de devolver o objeto atualizado).

• 401 Unauthorized: O cliente precisa se autenticar para obter a resposta solicitada 
  (falta de token ou credenciais inválidas).

• 403 Forbidden: O cliente foi autenticado, mas não tem direitos de acesso ao recurso 
  (permissão insuficiente).

• 409 Conflict: A requisição não pôde ser concluída devido a um conflito com o estado atual do 
  servidor (ex: tentar cadastrar um usuário com um e-mail que já existe).

package com.julyturadm.service;

import com.julyturadm.domain.entity.User;
import com.julyturadm.domain.repository.UserRepository;
import com.julyturadm.dto.request.UserCreateRequest;
import com.julyturadm.dto.response.UserResponse;

import lombok.RequiredArgsConstructor;

import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import com.julyturadm.common.exception.BusinessException;

import java.util.List;
import com.julyturadm.common.exception.ResourceNotFoundException;

@Service
@RequiredArgsConstructor
@Transactional
public class UserService {

    private final UserRepository userRepository;
    private final PasswordEncoder passwordEncoder;

    public UserResponse createUser(UserCreateRequest request) {

        if (userRepository.existsByEmail(request.email())) {
            throw new BusinessException(
                    "Já existe um usuário cadastrado com este e-mail.");
        }

        String encodedPassword = passwordEncoder.encode(request.password());

        User user = User.create(request.name(), request.email(), encodedPassword);

        User savedUser = userRepository.save(user);

        return toResponse(savedUser);
    }

    @Transactional(readOnly = true)
    public UserResponse findUserById(Long id) {
        User user = userRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + id));

        return toResponse(user);
    }

    @Transactional(readOnly = true)
    public UserResponse findUserByEmail(String email) {
        User user = userRepository.findByEmail(email)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o e-mail: " + email));

        return toResponse(user);
    }

    @Transactional(readOnly = true)
    public List<UserResponse> findAllActiveUsers() {
        return userRepository.findAllByEnabledTrue()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    public UserResponse disableUser(Long id) {
        User user = userRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + id));

        user.setEnabled(false);

        User updatedUser = userRepository.save(user);

        return toResponse(updatedUser);
    }

    public UserResponse enableUser(Long id) {
        User user = userRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + id));

        user.setEnabled(true);

        User updatedUser = userRepository.save(user);

        return toResponse(updatedUser);
    }
    
    private UserResponse toResponse(User user) {

        return new UserResponse(
                user.getId(),
                user.getName(),
                user.getEmail(),
                user.getEnabled(),
                user.getCreatedAt(),
                user.getUpdatedAt());
    }

}

```
package com.julyturadm.service;

import com.julyturadm.common.exception.BusinessException;
import com.julyturadm.common.exception.ResourceNotFoundException;
import com.julyturadm.domain.entity.Role;
import com.julyturadm.domain.entity.User;
import com.julyturadm.domain.entity.UserRole;
import com.julyturadm.domain.repository.RoleRepository;
import com.julyturadm.domain.repository.UserRepository;
import com.julyturadm.domain.repository.UserRoleRepository;
import com.julyturadm.dto.response.RoleResponse;

import lombok.RequiredArgsConstructor;

import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional
public class UserRoleService {

    private static final String ADMIN_ROLE = "ADMIN";

    private final UserRepository userRepository;
    private final RoleRepository roleRepository;
    private final UserRoleRepository userRoleRepository;

    public RoleResponse assignRole(Long userId, Long roleId) {

        User user = findUser(userId);
        Role role = findRole(roleId);

        validateRoleAssignment(role);

        if (userRoleRepository.existsByUser_IdAndRole_Id(userId, roleId)) {
            throw new BusinessException(
                    "Este perfil já está associado ao usuário.");
        }

        UserRole userRole = UserRole.create(user, role);

        userRoleRepository.save(userRole);

        return toResponse(role);
    }

    @Transactional(readOnly = true)
    public List<RoleResponse> findUserRoles(Long userId) {

        findUser(userId);

        return userRoleRepository.findAllByUser_Id(userId)
                .stream()
                .map(userRole -> toResponse(userRole.getRole()))
                .toList();
    }

    public void removeRole(Long userId, Long roleId) {

        findUser(userId);
        Role role = findRole(roleId);

        validateRoleAssignment(role);

        UserRole userRole = userRoleRepository
                .findById(new com.julyturadm.domain.entity.id.UserRoleId(
                        userId, roleId))
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Associação entre usuário e perfil não encontrada."));

        userRoleRepository.delete(userRole);
    }

    private void validateRoleAssignment(Role role) {

        Authentication authentication = SecurityContextHolder
                .getContext()
                .getAuthentication();

        if (authentication == null || !authentication.isAuthenticated()) {
            throw new AccessDeniedException(
                    "Usuário não autenticado.");
        }

        boolean isAdmin = authentication.getAuthorities().stream()
                .anyMatch(authority ->
                        authority.getAuthority().equals("ROLE_ADMIN"));

        boolean isRh = authentication.getAuthorities().stream()
                .anyMatch(authority ->
                        authority.getAuthority().equals("ROLE_RH"));

        if (isAdmin) {
            return;
        }

        if (isRh && !ADMIN_ROLE.equals(role.getName())) {
            return;
        }

        throw new AccessDeniedException(
                "Você não tem permissão para atribuir ou remover este perfil.");
    }

    private User findUser(Long userId) {
        return userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + userId));
    }

    private Role findRole(Long roleId) {
        return roleRepository.findById(roleId)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Perfil não encontrado com o ID: " + roleId));
    }

    private RoleResponse toResponse(Role role) {
        return new RoleResponse(
                role.getId(),
                role.getName(),
                role.getCreatedAt(),
                role.getUpdatedAt());
    }
}
```

    }

    // `findOrCreateRole()` e `findOrCreatePermission()` evitam criar registros duplicados.
    private Permission findOrCreatePermission(String name) {

        return permissionRepository.findByName(name)
                .orElseGet(() -> permissionRepository.save(Permission.create(name)));
    }

    // - `grantPermission()` verifica se a associação já existe antes de gravá-la.
    // - `RolePermission.create()` recebe entidades já persistidas, com seus IDs disponíveis.
    private void grantPermission(Role role, Permission permission) {

        boolean alreadyExists =
                rolePermissionRepository.existsByRole_IdAndPermission_Id(
                        role.getId(),
                        permission.getId()
                );

        if (!alreadyExists) {
            RolePermission association =
                    RolePermission.create(role, permission);

            rolePermissionRepository.save(association);
        }
    }
}
