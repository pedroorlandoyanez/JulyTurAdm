Para lembrar dos codigos HTTP combinados com os verbos do HTTP:

• 200 OK: A requisição foi bem-sucedida. O significado do sucesso depende do método HTTP utilizado
  (ex: em um GET, os dados solicitados são retornados no corpo da resposta).

• 204 No Content: A requisição foi processada com sucesso, mas não há conteúdo para retornar no
  corpo da resposta (altamente utilizado em requisições DELETE ou PUT/PATCH quando não há necessidade
  de devolver o objeto atualizado).

• 401 Unauthorized: O cliente precisa se autenticar para obter a resposta solicitada 
  (falta de token ou credenciais inválidas).

• 403 Forbidden: O cliente foi autenticado, mas não tem direitos de acesso ao recurso 
  (permissão insuficiente).

• 409 Conflict: A requisição não pôde ser concluída devido a um conflito com o estado atual do 
  servidor (ex: tentar cadastrar um usuário com um e-mail que já existe).

import jakarta.persistence.*;

package com.julyturadm.domain.entity;

import com.julyturadm.domain.entity.id.RolePermissionId;
import jakarta.persistence.*;
import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

@Entity
@Table(name = "roles_permissions")
@Getter
@Setter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class RolePermission {

    // - `RolePermissionId` recebe os IDs dos dois registros.
    @EmbeddedId
    private RolePermissionId id;

    // ---
    // - `setRole()` estabelece o relacionamento com o perfil.
    // - `@MapsId` mantém os campos da chave composta sincronizados com os
    //    relacionamentos no momento da persistência.
    // ---
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @MapsId("roleId")
    @JoinColumn(name = "role_id", nullable = false, foreignKey = @ForeignKey(name = "FK_rp_role"))
    private Role role;

    // ---
    // - `setPermission()` estabelece o relacionamento com a permissão.
    // - `@MapsId` mantém os campos da chave composta sincronizados com os
    //    relacionamentos no momento da persistência.
    // ---
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @MapsId("permissionId")
    @JoinColumn(name = "permission_id", nullable = false, foreignKey = @ForeignKey(name = "FK_rp_permission"))
    private Permission permission;
       
    // ---
    // Esse método permite criar uma role e uma permissao sem expor publicamente o 
    // construtor da entidade.
    // ---    
    public static RolePermission create(
            Role role,
            Permission permission) {

        RolePermission rolePermission = new RolePermission();

        rolePermission.setId(
                new RolePermissionId(
                        role.getId(),
                        permission.getId()));

        rolePermission.setRole(role);
        rolePermission.setPermission(permission);

        return rolePermission;
    }

}package com.julyturadm.domain.entity;

import com.julyturadm.domain.entity.id.UserRoleId;
import jakarta.persistence.*;
import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

@Entity
@Table(name = "users_roles")
@Getter
@Setter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class UserRole {

    @EmbeddedId
    private UserRoleId id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @MapsId("userId")
    @JoinColumn(name = "user_id", nullable = false, foreignKey = @ForeignKey(name = "FK_ur_user"))
    private User user;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @MapsId("roleId")
    @JoinColumn(name = "role_id", nullable = false, foreignKey = @ForeignKey(name = "FK_ur_role"))
    private Role role;

    public static UserRole create(User user, Role role) {
        UserRole userRole = new UserRole();

        userRole.setId(new UserRoleId(user.getId(), role.getId()));
        userRole.setUser(user);
        userRole.setRole(role);

        return userRole;
    }
}
