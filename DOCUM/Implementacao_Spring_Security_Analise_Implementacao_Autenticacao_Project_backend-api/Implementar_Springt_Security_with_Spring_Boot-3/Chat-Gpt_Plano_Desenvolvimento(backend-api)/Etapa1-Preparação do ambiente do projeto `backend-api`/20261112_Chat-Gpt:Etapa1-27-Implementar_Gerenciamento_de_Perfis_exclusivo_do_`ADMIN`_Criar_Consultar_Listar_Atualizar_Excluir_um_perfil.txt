Para lembrar dos codigos HTTP combinados com os verbos do HTTP:

• 200 OK: A requisição foi bem-sucedida. O significado do sucesso depende do método HTTP utilizado
  (ex: em um GET, os dados solicitados são retornados no corpo da resposta).

• 204 No Content: A requisição foi processada com sucesso, mas não há conteúdo para retornar no
  corpo da resposta (altamente utilizado em requisições DELETE ou PUT/PATCH quando não há necessidade
  de devolver o objeto atualizado).

• 401 Unauthorized: O cliente precisa se autenticar para obter a resposta solicitada 
  (falta de token ou credenciais inválidas).

• 403 Forbidden: O cliente foi autenticado, mas não tem direitos de acesso ao recurso 
  (permissão insuficiente).

• 409 Conflict: A requisição não pôde ser concluída devido a um conflito com o estado atual do 
  servidor (ex: tentar cadastrar um usuário com um e-mail que já existe).

package com.julyturadm.domain.repository;

import com.julyturadm.domain.entity.Role;
import org.springframework.data.jpa.repository.JpaRepository;

import java.util.Optional;

import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import java.util.List;

public interface RoleRepository extends JpaRepository<Role, Long> {

    Optional<Role> findByName(String name);

    boolean existsByName(String name);

    //---
    // `JOIN r.userRoles ur`: identifica os papéis associados ao usuário.
    // `LEFT JOIN FETCH r.rolePermissions`: carrega as permissões desses papéis.
    // `LEFT JOIN FETCH rp.permission`: carrega os dados de cada permissão.
    //---
    @Query("""
        SELECT DISTINCT r
        FROM Role r
        JOIN r.userRoles ur
        LEFT JOIN FETCH r.rolePermissions rp
        LEFT JOIN FETCH rp.permission
        WHERE ur.user.email = :email
    """)
    List<Role> findAllByUserEmailWithPermissions(
    @Param("email") String email
    );
}

package com.julyturadm.service;

import com.julyturadm.domain.entity.User;
import com.julyturadm.domain.repository.UserRepository;
import com.julyturadm.dto.request.UserCreateRequest;
import com.julyturadm.dto.response.UserResponse;

import lombok.RequiredArgsConstructor;

import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import com.julyturadm.common.exception.BusinessException;

import java.util.List;
import com.julyturadm.common.exception.ResourceNotFoundException;

@Service
@RequiredArgsConstructor
@Transactional
public class UserService {

    private final UserRepository userRepository;
    private final PasswordEncoder passwordEncoder;

    public UserResponse createUser(UserCreateRequest request) {

        if (userRepository.existsByEmail(request.email())) {
            throw new BusinessException(
                    "Já existe um usuário cadastrado com este e-mail.");
        }

        String encodedPassword = passwordEncoder.encode(request.password());

        User user = User.create(request.name(), request.email(), encodedPassword);

        User savedUser = userRepository.save(user);

        return toResponse(savedUser);
    }

    @Transactional(readOnly = true)
    public UserResponse findUserById(Long id) {
        User user = userRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + id));

        return toResponse(user);
    }

    @Transactional(readOnly = true)
    public UserResponse findUserByEmail(String email) {
        User user = userRepository.findByEmail(email)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o e-mail: " + email));

        return toResponse(user);
    }

    @Transactional(readOnly = true)
    public List<UserResponse> findAllActiveUsers() {
        return userRepository.findAllByEnabledTrue()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    public UserResponse disableUser(Long id) {
        User user = userRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + id));

        user.setEnabled(false);

        User updatedUser = userRepository.save(user);

        return toResponse(updatedUser);
    }

    public UserResponse enableUser(Long id) {
        User user = userRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Usuário não encontrado com o ID: " + id));

        user.setEnabled(true);

        User updatedUser = userRepository.save(user);

        return toResponse(updatedUser);
    }
    
    private UserResponse toResponse(User user) {

        return new UserResponse(
                user.getId(),
                user.getName(),
                user.getEmail(),
                user.getEnabled(),
                user.getCreatedAt(),
                user.getUpdatedAt());
    }

}

//- Usuário autenticado sem `USER_CREATE` não poderá criar usuários.
//- Usuário com `USER_READ` poderá consultar usuários.
//- Usuário sem `USER_DISABLE` não poderá desativar contas.
//- Usuário sem `USER_ENABLE` não poderá reativar contas.
//- Requisições sem autenticação continuarão sendo rejeitadas.

@RestController
@RequestMapping("/api/users")
@RequiredArgsConstructor
public class UserController {

    private final UserService userService;

    private final UserRoleService userRoleService;

    @PostMapping
    @PreAuthorize("hasAuthority('USER_CREATE')")
    public ResponseEntity<UserResponse> createUser(
            @Valid @RequestBody UserCreateRequest request) {

        UserResponse response = userService.createUser(request);

        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(response);
    }

    @GetMapping("/{id}")
    @PreAuthorize("hasAuthority('USER_READ')")
    public ResponseEntity<UserResponse> findUserById(
            @PathVariable Long id) {

        return ResponseEntity.ok(
            userService.findUserById(id)
        );
    }

    @GetMapping("/email/{email}")
    @PreAuthorize("hasAuthority('USER_READ')")
    public ResponseEntity<UserResponse> findUserByEmail(
            @PathVariable String email) {

        return ResponseEntity.ok(
            userService.findUserByEmail(email)
        );
    }

    @GetMapping
    @PreAuthorize("hasAuthority('USER_READ')")
    public ResponseEntity<List<UserResponse>> findAllActiveUsers() {

        return ResponseEntity.ok(
            userService.findAllActiveUsers()
        );
    }

    @PatchMapping("/{id}/disable")
    @PreAuthorize("hasAuthority('USER_DISABLE')")
    public ResponseEntity<UserResponse> disableUser(
            @PathVariable Long id) {

        return ResponseEntity.ok(
            userService.disableUser(id)
        );
    }

    @PatchMapping("/{id}/enable")
    @PreAuthorize("hasAuthority('USER_ENABLE')")
    public ResponseEntity<UserResponse> enableUser(
            @PathVariable Long id) {

        return ResponseEntity.ok(
            userService.enableUser(id)
        );
    }

    //---
    // Por que usamos duas verificações de segurança?
    // 1. @PreAuthorize no controller permite que somente ADMIN e RH acessem as operações

    // 2. validateRoleAssignment() no UserRoleService verifica o perfil do usuário autenticado
    //    e impede especificamente que RH atribua ou remova ADMIN.
    //---
    @GetMapping("/{userId}/roles")
    @PreAuthorize("hasAuthority('USER_READ')")

        return ResponseEntity.ok(

            @PathVariable Long userId,
            @PathVariable Long roleId) {

        return ResponseEntity.ok(
            userRoleService.assignRole(userId, roleId)

    }

    // 
    @DeleteMapping("/{userId}/roles/{roleId}")
    @PreAuthorize("hasAnyRole('ADMIN', 'RH')")
    public ResponseEntity<Void> removeRole(
            @PathVariable Long userId,
            @PathVariable Long roleId) {

        userRoleService.removeRole(userId, roleId);

        return ResponseEntity.noContent().build();
    }    
}

```
package com.julyturadm.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.julyturadm.domain.entity.Role;
import com.julyturadm.domain.repository.RoleRepository;
import com.julyturadm.domain.repository.RolePermissionRepository;
import com.julyturadm.domain.repository.UserRoleRepository;
import com.julyturadm.dto.request.RoleCreateRequest;
import com.julyturadm.dto.request.RoleUpdateRequest;
import com.julyturadm.dto.response.RoleResponse;
import com.julyturadm.exception.BusinessException;
import com.julyturadm.exception.ResourceNotFoundException;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class RoleService {

    private final RoleRepository roleRepository;
    private final UserRoleRepository userRoleRepository;
    private final RolePermissionRepository rolePermissionRepository;

    public RoleResponse createRole(RoleCreateRequest request) {

        if (roleRepository.existsByName(request.name())) {
            throw new BusinessException(
                    "Já existe um perfil com esse nome.");
        }

        Role role = Role.create(request.name());

        Role savedRole = roleRepository.save(role);

        return toResponse(savedRole);
    }

    @Transactional(readOnly = true)
    public RoleResponse findRoleById(Long id) {

        Role role = findRoleOrThrow(id);

        return toResponse(role);
    }

    @Transactional(readOnly = true)
    public List<RoleResponse> findAllRoles() {

        return roleRepository.findAll()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    public RoleResponse updateRole(Long id, RoleUpdateRequest request) {

        Role role = findRoleOrThrow(id);

        if (roleRepository.existsByNameAndIdNot(request.name(), id)) {
            throw new BusinessException(
                    "Já existe outro perfil com esse nome.");
        }

        role.setName(request.name());

        Role updatedRole = roleRepository.save(role);

        return toResponse(updatedRole);
    }

    public void deleteRole(Long id) {

        Role role = findRoleOrThrow(id);

        if (userRoleRepository.existsByRole_Id(id)) {
            throw new BusinessException(
                    "Não é possível excluir o perfil: ele está associado a usuários.");
        }

        if (rolePermissionRepository.existsByRole_Id(id)) {
            throw new BusinessException(
                    "Não é possível excluir o perfil: ele possui permissões associadas.");
        }

        roleRepository.delete(role);
    }

    private Role findRoleOrThrow(Long id) {

        return roleRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Perfil não encontrado: " + id));
    }

    private RoleResponse toResponse(Role role) {

        return new RoleResponse(
                role.getId(),
                role.getName(),
                role.getCreatedAt(),
                role.getUpdatedAt());
    }
}
```

package com.julyturadm.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.julyturadm.domain.entity.Role;
import com.julyturadm.domain.repository.RoleRepository;
import com.julyturadm.domain.repository.RolePermissionRepository;
import com.julyturadm.domain.repository.UserRoleRepository;
import com.julyturadm.dto.request.RoleCreateRequest;
import com.julyturadm.dto.request.RoleUpdateRequest;
import com.julyturadm.dto.response.RoleResponse;
import com.julyturadm.common.exception.BusinessException;
import com.julyturadm.common.exception.ResourceNotFoundException;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class RoleService {

    private final RoleRepository roleRepository;
    private final UserRoleRepository userRoleRepository;
    private final RolePermissionRepository rolePermissionRepository;

    public RoleResponse createRole(RoleCreateRequest request) {

        if (roleRepository.existsByName(request.name())) {
            throw new BusinessException(
                    "Já existe um perfil com esse nome.");
        }

        Role role = Role.create(request.name());

        Role savedRole = roleRepository.save(role);

        return toResponse(savedRole);
    }

    @Transactional(readOnly = true)
    public RoleResponse findRoleById(Long id) {

        Role role = findRoleOrThrow(id);

        return toResponse(role);
    }

    @Transactional(readOnly = true)
    public List<RoleResponse> findAllRoles() {

        return roleRepository.findAll()
                .stream()
                .map(this::toResponse)
                .toList();
    }

    public RoleResponse updateRole(Long id, RoleUpdateRequest request) {

        Role role = findRoleOrThrow(id);

        if (roleRepository.existsByNameAndIdNot(request.name(), id)) {
            throw new BusinessException(
                    "Já existe outro perfil com esse nome.");
        }

        role.setName(request.name());

        Role updatedRole = roleRepository.save(role);

        return toResponse(updatedRole);
    }

    public void deleteRole(Long id) {

        Role role = findRoleOrThrow(id);

        if (userRoleRepository.existsByRole_Id(id)) {
            throw new BusinessException(
                    "Não é possível excluir o perfil: ele está associado a usuários.");
        }

        if (rolePermissionRepository.existsByRole_Id(id)) {
            throw new BusinessException(
                    "Não é possível excluir o perfil: ele possui permissões associadas.");
        }

        roleRepository.delete(role);
    }

    private Role findRoleOrThrow(Long id) {

        return roleRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Perfil não encontrado: " + id));
    }

    private RoleResponse toResponse(Role role) {

        return new RoleResponse(
                role.getId(),
                role.getName(),
                role.getCreatedAt(),
                role.getUpdatedAt());
    }
}
